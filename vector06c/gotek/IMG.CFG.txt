# Для использования данной опции, скопируйте данный файл в корень
#   или директорию FF\ на USB Flash накопитель.
# ВНИМАНИЕ: если на носителе присутствует директория FF\ файл необходимо
#           разместить только в ней, а не в корневом каталоге!

# Правила применения:
#  0. Секция описаний геометрии (далее "тэг") начинается с имени,
#     и размера файла, раздел нные "::" и заключенные в квадратные скобки.
#     [<имятэга>::<размер файла в байтах>]
#         или
#     [<имятэга>]
#         или
#     [::<размер файла в байтах>]
#     Данная строка помещаются в начало секции.
#     Например [sssd40::92160]
#  1. Если определ н тэг [<имятэга>], описание геометрии, определ нное
#     под этим тэгом, будет применено ко всем файлам имиджей с именами
#     вида: *.<имятэга>.img и *.<имятэга>.ima
#     Например system1.sssd40.img
#  2. Если определ н тэг со пустым именем [::<размер файла в байтах>],
#     описание геометрии, соответствующее данному тэгу, будет применено
#     ко всем файлам, имена которых не попадают под шаблон иных определ нных
#     тэгов (таких, как *.img и *.ima).
#     Например userdat2.img
#  3. Имена тэгов и соответствующие им шаблоны НЕЧУВСТИВИТЕЛЬНЫ К РЕГИСТРУ
#  4. Значение поля <размер файла в байтах>, если оно присутствует в описании
#     тэга должно точно соответствовать размеру файлов имиджа.
#  5. Если имя какого-либо файла имиджа не подходит ни под один из шаблонов
#     определ нных тэгов, используется геометрия по умолчанию, соответствующая
#     типу хост-устройства, определ нного в FF.CFG (host=<тип хост-платформы>)

#   приводит к генерации сообщения error 31 (Bad Image/Неправильный образ).

# шаблону *.dsdd80.{img,ima,dsk}
# и имеющих размер равный 720кБ (737280 байт).
#[dsdd80::737280]

## ПАРАМЕТРЫ ДИСКА
#  Применяются для всего имиджа, за исключением дорожек, описанных
#  в субсекции "tracks"

# ОБЯЗАТЕЛЬНЫЙ ПАРАМЕТР: Количество цилиндров (1-255).
# cyls = 84

# ОБЯЗАТЕЛЬНЫЙ ПАРАМЕТР: Количество головок (1-2).
# heads = 2

# Опциональный параметр: Количество шагов головок, которые необходмио
# сделать для перемещения на следующий цилиндр диска. (Например, для
# симуляции 40 дорожечной дискеты на 80-дорожечном накопителе)
# Значение по умолчанию 1
# step = 1

# Опциональный параметр: Порядок следования дорожек в файле имиджа
#   sequential:    Последовательная нумерация цилиндров - сначала все
#                   дорожки стороны 0, затем стороны 1.
#   interleaved:   Расположение дорожек с чередованием (поцилиндрово):
#                  дор0/стор0, затем дор0/стор1, дор1/стор0, дор1/стор1,
#                  и т.д. (используется по умолчанию).
#   reverse-sideN: Реверсировать порядок цилиндров для стороны N (N=0,1)
#                   начиная от центра к краю.
#   sides-swapped: Изменить порядок сторон на противоположный.
# Может быть указано несколько значений, разделяемых запятой,
#   например: file-layout = sequential,reverse-side1
#file-layout = interleaved

## ПАРАМЕТРЫ ДОРОЖКИ
# Параметры могут отличаться для различных дорожек.
# По умолчанию они применяются ко всем дорожкам, за исключением тех,
# которые специально описаны с помощью объявления "tracks".

# tracks = <track-list>
# Определяет дорожки с особыми параметрами
# Format:
#  track-list  : <диапазон>[,<диапазон>[,<диапазон>...]]
#  диапазон    : <цилиндр>[.<головка>] | <нач.цилиндр>-<кон.цилиндр>[.<головка>]
#  цилиндр     : [0-9][0-9]*
#  головка     : 0 | 1
# Если не указан номер головки, применяется для обеих головок (0 и 1)
# Примеры:
#  tracks = 0-23.1,45.0
#   - определяет цилиндры в диапазоне 0-23/головка 1 + цилиндр 45/головка 0
#  tracks = 27-33
#   - определяет цилиндры в диапазоне 27-33, применительно к обеим головкам.

# Опциональный параметр: Количество секторов на дорожку (Sectors per track).
# Допустимый диапазон: 1-256. По умолчанию: 0
# secs = 5

# ОБЯЗАТЕЛЬНЫЙ ПАРАМЕТР (если параметр "secs" не равен 0):
# Размер сектора в байтах (Bytes per sector)
# Допустимые значения: 128, 256, 512, 1024, 2048, 4096, 8192.
# bps = 1024

# Опциональный параметр: Идентификатор (ID) 1-го сектора на дорожке
# Значения могут быть указаны в шестнадцатиричном виде в формате 0x<число>
# Допустимый диапазон: 0-255
  id = 1

# Опциональный параметр: номер головки для всех секторов дорожки.
# Допустимые значения [auto|0|1]
#     auto: использовать текущий физический номер головки (по умолчанию).
  h = auto

# Опциональный параметр: Метод записи (fm | mfm).
# Значение по умолчанию mfm.
  mode = mfm

# Опциональный параметр: Чередование секторов (Sector interleave).
# Значение по умолчанию 1:1 (без чередования).
# interleave = 1

# Опциональный параметр: Смещение секторов по цилиндрам
#  (Sector skew per cylinders).
# Значение по умолчанию 0 (без смещения).
 cskew = 4

# Опциональный параметр: Смещение секторов по головкам
#  (Sector skew per head).
# Значение по умолчанию 0 (без смещения).
# hskew = 4

# Опциональный параметр: Скорость вращения диска (об/м)
#  (Rotational RPM) (честота гъвкавия).
# Значение по умолчанию 300.
# rpm = 300

# Опциональный параметр: Величина зазора после идентификатора
# (Post-ID Gap)
# Допустимые значения: [auto|0..255].
#     auto: использовать значение в соответствии с размером сектора
#           и методом записи (по умолчанию).
#     auto = based on recording mode and sector size.
# gap2 = 22

# Опциональный параметр: Величина зазора после поля данных
#  (Post-Data Gap)
# Допустимые значения: [auto|0..255].
#     auto: использовать значение в соответствии с размером сектора
#           и методом записи (по умолчанию).
# gap3 = 116

# Опциональный параметр: Величина зазора после индексного поля
#  (Post-Index Gap)
# Допустимые значения: [auto|0..255].
#     auto: использовать значение в соответствии с размером сектора
#           и методом записи (по умолчанию).
# gap4B = 210


# Опциональный параметр: Наличие индексного маркера
#  (Index Address Mark) (yes | no).
# Значение по умолчанию yes
  iam = yes

# Опциональный параметр: Скорость передачи данных (кГц) (кбит/с)
#  (Data rate) (например MFM DD = 250).
#  Значение = 0: Скорость передачи определяется согласно методу
#                записи и размеру дорожки (по умолчанию).
# Значение по умолчанию 0
  rate = 250

# Пример определения образов для TSC Flex.
#
# Flex использует формат двойной плотности (double-density),
# но цилиндр 0 записывается с одинарную плотностью (single-density FM)
# Нумерация секторов и фактор чередования на диске непрерывно меняется.
# Поскольку нет прямого способа реализации, реализуем это следующим образом:
# Укажем последовательность идентификации секторов для стороны 1 после
# дорожек стороны 0 с помощью параметра "id"
# Далее, с помощью параметра "hskew" смещаем 1-й сектор стороны 1
# на необходимую величину, для имитации перекрестного чередования секторов
# в зависимости от дорожки.

#[flex::733184]
#cyls = 80
#heads = 2
#bps = 256       # Основная масса секторов 256 байт
#tracks = 0.0    ## Загрузочный цилиндр, головка 0
 # secs = 10
 # mode = fm
  #interleave = 4
 # id = 1
#tracks = 0.1    ## Загрузочный цилиндр, головка 1
 # secs = 10
 # mode = fm
  #interleave = 4
 # hskew = 1     # Иммитируем правильное чередование между дорожками
 # id = 11       # Идентификаторы начинаются с 11 (следом за головкой 0: 1..10)
#tracks = 1-79.0 ## Для всех остальных цилиндров, головка 0
 # secs = 18
 # mode = mfm
 # interleave = 6
 # id = 1
#tracks = 1-79.1 ## Для всех остальных цилиндров, головка 1
#  secs = 18
  #mode = mfm
 # interleave = 6
  #hskew = 5     # Иммитируем правильное чередование между дорожками
 # id = 19       # Идентификаторы начинаются с 19 (следом за головкой 0: 1..18)
