#  параметрами в других файлах конфигурации.

#  файл рекомендуется удалить с рабочего USB Flash накопителя.

## ЭМУЛЯЦИЯ FLOPPY НАКОПИТЕЛЯ

# Режим работы интерфейса Floppy накопителя (управление линиями 2 и 34)
# jc: управление с помощью перемычки JC:
# установлена - интерфейс IBM PC, снята - Shugart
# shugart: линия P2=DSKCHG, линия P34=RDY (Amiga, Atari ST, УК-НЦ, и др...)
# ibmpc: линия P2 не используется, линия P34=DSKCHG (интерфейс IBM PC)
# ibmpc-hdout: P2=HD_OUT, P34=DSKCHG (обычно устанваливать данный параметр
#              необходимости нет, желательно использование параметра 'ibmpc')
# jppc: линия P2 не используется, линия P34=RDY (Японский стандарт PC)
# jppc-hdout: P2=HD_OUT, P34=RDY (Альтернативный вариант японского стандарта PC
#             желательно использования параметра 'jppc')
# akai-s950: Устаревший алиас 'jppc-hdout' использовался для Akai S950
# amiga: линия P2=DSKCHG, линия P34=DRIVE_ID (обычно необходимости нет,
#              предпочтительно использования параметра 'shugart')
  interface = shugart

# Платформа внешнего ПО: Позволяет правильно определить формат файлов имиджа
#   для многоформатных типов, напрмиер таких, как .IMG
# acorn: Acorn ADFS
# akai: Akai (S01, S20, S950), Korg, SC Prophet 3000
# casio: Casio (FZ-1)
# dec: DEC (RX33, RX50)
# ensoniq: Ensoniq (ASR, TS, etc)
# fluke: Fluke 9100
# gem: General Music (S2, S3)
# ibm-3174: IBM 3174 Establishment Controller
# memotech: Memotech
# msx: MSX
# nascom: Nascom
# pc98: NEC PC-98
# pc-dos: Формат PC DOS (установки определяются из BIOS)
# tandy-coco: Tandy Color Computer (CoCo)
# ti99: TI-99/4A
# uknc: УК-НЦ / ДВК (MZ/MY) - советский клон PDP-11
# unspecified: Общий формат по умолчанию (включая IBM PC)
  host = pc-dos

# Ручное определение функций выходных линий 2 и 34 Floppy интерфейса
# auto: Конфигурирование по умолчанию, согласно установленному
#       параметру "interface"
# nc: Не используется/не подключена
# low: Всегда установлен В 0 (0v)
# high: Всегда установлено +5 Вольт (5v)
# rdy: Функция "Drive ready" (активен/Ready = 0v)
# nrdy: То же, что и "rdy", но с инверсным уровнем
# dens: Индикатор плотности записи / Density mode (High Density = 0v)
# ndens: То же, что и "dens", но с инверсным уровнем
# chg: Индикатор отсутствия диска в накопителе / Disk changed (Changed = 0v)
# nchg: То же, что и "chg", но с инверсным уровнем
# Допустимые значения: auto|nc|low|high|rdy|nrdy|dens|ndens|chg|nchg
  pin02 = auto
  pin34 = auto

# Установить принудительно защиту записи для имиджей?
#  (иначе в силу вступает атрибут "read-only" для файла имиджа)
# Допустимые значения: yes | no
  write-protect = no

# Максимальное кол-во цилиндров для эмклируемой дискеты.
# Значение 255 требуется для доступа в режиме "Direct Access", используемого в
# утилитах работающих с образами в режиме прямого доступа, а также игр,
# использующих функции автоматической смены диска)
# Допустимый диапазон значений: 0 <= N <= 255
  max-cyl = 255

# Фильтр коротких импульсов сигнала "SIDE-select" / "Выбор стороны"
#  подавляются импульсы менее, чем N микроскеунд
# Допустимый диапазон значений: 0 <= N <= 255
#side-select-glitch-filter = 90

# Управление эмуляцией вращения диска при смене дорожки
# instant: При смене дорожки эмуляция вращения приостанавливается
# realtime: Вращение диска всегда эмулируется, даже при смене дорожки.
# Допустимые значения: instant | realtime
  track-change = realtime

# Управление эмуляцией вращения диска после окончания записи во Flash
# instant: Эмуляция вращения приостанавливается
# realtime: Эмуляция вращения не прерывается
# eot: Эмулируемая дискета "перемещается" на конец дорожки
# Допустимые значения: instant | realtime | eot
  write-drain = realtime

# Подавлять ли индексные импульсы (Index),
# если неактивны сигналы RDATA и WDATA?
# Допустимые значения: yes | no
  index-suppression = yes

# Время в милисекундах от начала перемещения головки на шаг
# до активизации сигнала RDATA
# Допустимый диапазон значений: 0 <= N <= 255
  head-settle-ms = 3

# Задержка в милисекундах от команды на включения мотора
# до установки готовности привода
# ПРИМЕЧАНИЕ: Обратитесь к поставщику оборудования для уточнения
#             доступности данной опции.
# Допустимый диапазон значений: ignore | 0 <= N <= 1000
  motor-delay =3

# Что является условием для деактивации сигнала disk-change (chg) после
# смены/установки диска
# step: Принята команда "шаг" (Step)
# pa14: Сигнал CHGRST (интерфейсная линия 1 на старых приводах Sony),
# подключ нный к выводу PA14 (JTCK/SWCLK) МП (обратитесь к поставщику)
# delay-N: Автоматичеки по истечении времени N*0.5 секунд (0 <= N <= 15)
#chgrst = step

## ЗАПУСК / ИНИЦИАЛИЗАЦИЯ

# Иммитировать извлечение носителя при запуске?
#  (иначе сразу происходит загрузка имиджа)
# Допустимые значения: yes | no
  ejected-on-startup = no

# Какой файл имиджа (или каталог) выбирать при запуске.
# last: Последний на момент выключения питания (сохраняется в IMAGE_A.CFG)
# static: Постоянно выбранный, указанный в файле INIT_A.CFG
# init: Первый в корневом каталоге
# Допустимые значения: last | static | init
  image-on-startup = static

# Время в милисекундах, отводимое на попытку определения
#  подключ нного индикатора. Если используется 2х символьный LED индикатор,
#  допустимо устанавливать значение параметра в 0
# Допустимый диапазон значений: 0 <= N <= 65535
  display-probe-ms = 3000
# ВНИМАНИЕ: ПОЖАЛУЙСТА НЕ ИЗМЕНЯЙТЕ ДАННЫЙ ПАРАМЕТР.

## НАВИГАЦИЯ ПО ФАЙЛАМ ИМИДЖЕЙ

# Автоматически открывать/монтировать текущий файл после N секунд
# N=0: Блокировано автоматическое открытие файла
# Допустимый диапазон значений: 0 <= N <= 255
  autoselect-file-secs = 2

# Автоматически выбирать текущий каталог после N секунд
# N=0: Блокирован автоматический выбор каталога
# Допустимый диапазон значений: 0 <= N <= 255
  autoselect-folder-secs = 2

# Сортировка каталогов/субдиректорий для отображаемого (native)
# режима навигации
# always: Сортировка производится всегда.
# never: Сортировка не производится. Отображение согласно порядку в FAT.
# small: Частичная сортировка. Сортируются малые каталоги.
# Допустимые значения: always | never | small
  folder-sort = always

# Приоритет файлов/директорий при сортировке.
# folders: Сначала отображаются каталоги, после - файлы.
# files: Сначала отображаются файлы, каталоги - после.
# none: Между файлами и каталогами нет разницы при сортировке.
# Допустимые значения: folders | files | none
  sort-priority = folders

# Режим навигации для выбора имиджей/слотов
# native:  Навигация по всем допустимым имиджам/каталогам
# indexed: Навигация только по файлам имиджей с фиксированными именами:
#          DSKA0000, DSKA0001, и т.д. ... (см. параметр "indexed-prefix")
# default: Режим навигации, определяемый в режиме совместимости с HxC
#          соответствующей конфигурацией, в остальных случаях - "native"
# Допустимые значения: native | indexed | default
  nav-mode = default

# Выполнять ли циклическую навигацию по каталогу или слотам
#  между минимумом и максимумом?
# Допустимые значения: yes | no
  nav-loop = yes

# Функции двух основных кнопок
#              B1     | B2     | Обе
# zero:        Пред.  | След.  | Slot 0
# eject:       Пред.  | След.  | Извлечь/загрузить носитель
# htu:         +10    | +1     | +100
# rotary:      Вверх  | Выбор/Извлечь/Загрузить | -
# rotary-fast: Пред.  | След.  | На уровень вверх [Пред/След. ускоряются]
# reverse:     Функции кнопок B1 и B2 иеняются местами
# Допустимые значения: zero | eject | rotary | rotary-fast | reverse
# Допускается указывать несколько значений, раздел нных запятой,
# например: twobutton-action=eject,reverse
  twobutton-action = zero

# Настройка свойств и чувствительности поворотного энкодера.
# Закажите установку энкодера у поставщика оборудования.
# full: Один импульс на щелчок
# half: Два импульса на щелчок
# quarter: Четыре импульса на щелчок
# trackball: (*) Трекбол Blackberry-style (например на базе датчика Холла)
# buttons:   (*) Дополнительные кнопки Prev/Next
# reverse: Изменение направления вращения на противоположное
# none: Отсутствует
# Допустимые значения: none|quarter|half|full|trackball|buttons|reverse
# Допускается указывать несколько значений, раздел нных запятой,
# например: rotary=quarter,reverse
# Примечание: Опции, отмеченные (*) В настоящий момент не поставляются.
  rotary = full

# Префикс для им н файлов-образов для индексного (indexed)
# режима навигации. Строка также может быть пустой ("").
  indexed-prefix = "DSKA"

## ИНДИКАТОР

# Тип индикатора.
# auto: Автоопределение типа индикатора.
# oled-128xNN: I2C OLED 128 точек по ширине x NN по высоте (NN = 32 | 64)
#  -rotate: Для OLED поворот изображения на 180 градусов
#  -hflip:  Для OLED поворот изображения на 90 градусов
#  -inverse:    Инверсия изображения (ч рный текст на белом фоне)
# Допустимые значения: auto | oled-128xNN[-rotate|-hflip|-inverse]
  display-type = auto
# ВНИМАНИЕ: ПОЖАЛУЙСТА НЕ ИЗМЕНЯЙТЕ ДАННЫЙ ПАРАМЕТР БЕЗ КРАЙНЕЙ НЕОБХОДИМОСТИ.

# Выбор шрифта для OLED дисплея.
# Преимущество узкого   6x13 фонта:  Больше символов на строку
# Преимущество широкого 8x16 фонта:  Крупные буквы лучше видно!
# Допустимые значения: 6x13 | 8x16
  oled-font = 8x16

# Контрастность OLED дисплея.
# Допустимый диапазон значений: 0 <= N <= 255
  oled-contrast = 143

# Расположение информации и высота текста на дисплее / в режиме OSD.
# В качестве параметра может быть указан явный список отображения данных
# для каждой строки, начиная с верхней. Элементы списка разделяются запятыми.
# Каждый элемент списка - цифра, обозначающая представляемую информацю.
# Необязательный символ 'd' после цмфры - двойная высота строки (32-точки).
# Цифровые обозначения представляемой информации:
#    0: Имя текущего образа
#    1: Статусная строка
#    2: Информация о томе/образе
#    3: Имя текущей поддиректории
#    7: Пустая строка
# При установке параметра: display-order = default
# представление информации на экране эквивалентно display-order = 0,1
# Допустимые значения: default | [0-3,7][d]...
  display-order = default
  osd-display-order = default

# Кол-во текстовых колонок для OSD. Работают только при отключ нном LCD/OLED
#  *** В данной комплектации параметр не используется. Значение не менять.
# Допустимый диапазон значений: 16 <= N <= 40
  osd-columns = 40

# Отключение индикатора после N секунд бездействия
# N=0: всегда отключ н; N=255: всегда включ н
# Допустимый диапазон значений: 0 <= N <= 255
  display-off-secs = 255

# Когда включать индикатор после отключения
# yes: Включение при смене дорожки и операции записи
# sel: Включение при выборе устройства
# no:  Не включать автоматически
# Допустимые значения: yes | sel | no
  display-on-activity = yes

# Скролинг длинных им н файлов (для LCD/OLED) в милисекундах на обновление
# Допустимый диапазон значений: 100 <= N <= 65535
  display-scroll-rate = 200

# Время паузы в начале/окончании скролинга (для LCD/OLED) в милисекундах
# N=0: непрерывный скроллинг
# Допустимый диапазон значений: 0 <= N <= 65535
  display-scroll-pause = 2000

# Скролинг длинных им н файлов (для LCD/OLED) в милисекундах на обновление
# в процессе навигации
# Допустимый диапазон значений: 0 <= N <= 65535
  nav-scroll-rate = 80

# Время паузы перед началом скролинга (для LCD/OLED) в милисекундах
# в процессе навигации
# Допустимый диапазон значений: 0 <= N <= 65535
  nav-scroll-pause = 300
